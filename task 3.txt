class Solution {
    public int[] sortedSquares(int[] nums) {
    int n = nums.length;
        int[] ans = new int[n];
        int l = 0, r = n - 1;

        for (int i = n - 1; i >= 0; i--) {
            if (Math.abs(nums[l]) > Math.abs(nums[r])) {
                ans[i] = nums[l] * nums[l];
                l++;
            } else {
                ans[i] = nums[r] * nums[r];
                r--;
            }
        }

        return ans;    
    }
}


Input:
nums =[-4,-1,0,3,10]
Output:
[0,1,9,16,100]
