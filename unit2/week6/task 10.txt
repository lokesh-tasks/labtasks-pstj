import java.io.*;
import java.util.*;

public class Solution {

    static int n;

    static int[] manacherOdd(String s) {
        int m = s.length();
        int[] d = new int[m];

        int l = 0, r = -1;

        for (int i = 0; i < m; i++) {
            int k = (i > r) ? 1 : Math.min(d[l + r - i], r - i + 1);

            while (i - k >= 0 && i + k < m &&
                   s.charAt(i - k) == s.charAt(i + k)) {
                k++;
            }

            d[i] = k;

            if (i + k - 1 > r) {
                l = i - k + 1;
                r = i + k - 1;
            }
        }

        return d;
    }

    static int[] manacherEven(String s) {
        int m = s.length();
        int[] d = new int[m];

        int l = 0, r = -1;

        for (int i = 0; i < m; i++) {
            int k = (i > r) ? 0 :
                    Math.min(d[l + r - i + 1], r - i + 1);

            while (i - k - 1 >= 0 && i + k < m &&
                   s.charAt(i - k - 1) == s.charAt(i + k)) {
                k++;
            }

            d[i] = k;

            if (i + k - 1 > r) {
                l = i - k;
                r = i + k - 1;
            }
        }

        return d;
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br =
                new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();

        String t = s + s;

        int[] odd = manacherOdd(t);
        int[] even = manacherEven(t);

        int[] answer = new int[n];

        /*
         * For every rotation, calculate the longest
         * palindrome using palindrome centers.
         */
        for (int start = 0; start < n; start++) {

            int left = start;
            int right = start + n - 1;

            int best = 1;

            // Odd length palindromes
            for (int center = left; center <= right; center++) {

                int radius = odd[center];

                int maxLeft = center - left + 1;
                int maxRight = right - center + 1;

                radius = Math.min(radius,
                        Math.min(maxLeft, maxRight));

                best = Math.max(best, 2 * radius - 1);
            }

            // Even length palindromes
            for (int center = left; center <= right; center++) {

                int radius = even[center];

                int maxLeft = center - left;
                int maxRight = right - center + 1;

                radius = Math.min(radius,
                        Math.min(maxLeft, maxRight));

                best = Math.max(best, 2 * radius);
            }

            answer[start] = best;
        }

        StringBuilder out = new StringBuilder();

        for (int x : answer) {
            out.append(x).append('\n');
        }

        System.out.print(out);
    }
}
