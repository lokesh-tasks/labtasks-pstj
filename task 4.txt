class Solution {
    public boolean containsDuplicate(int[] nums) {
        Scanner sc = new Scanner (System.in);
        HashSet<Integer> set = new HashSet<>();
        for (int num : nums) {
            if (set.contains(num)) {
                return true;
            }
            set.add(num);
        }
        return false;
    }
}


Input:

nums =[1,2,3,1]

Output:

true
